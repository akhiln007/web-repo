def happyBirthday(person):

    print("Happy Birthday to you!")

    print("Happy Birthday to you!")

    print("Happy Birthday, dear " + person + ".")

    print("Happy Birthday to you!")

def main():

    happyBirthday('Akhil')

    happyBirthday('Nikhil')

main()
