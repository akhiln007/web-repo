l = [i ** 2 for i in range (1, 11)]

print(l)

print(l[2:9:2])

list = ['a', 'b', 'c', 'd', 'e', 'f'] 

print(list[1:3])

print(list[:4])

print(list[3:])

print(list[:])

letters=['A', 'B', 'C', 'D', 'E']

print(letters[ : : -1])
