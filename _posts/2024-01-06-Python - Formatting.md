#Strings

print('Hello, world!')

print('%s' % 'Hello, world!')

print('%20s' % 'Hello, world!')

print('%-20s' % 'Hello, world!')

print('%15s' % 'Hello, world!')

#Integers

print('%d'%12,345)

print('%d'%-12,345)

print('%20d'%12,345)

print('%20d'%-12,345)

print('%+d'%12,345)

print('%+d'%-12,345)

print('%+20d'%12,345)

print('%+020d'%-12,345)

#Floating

print('%f'%12.345)

print('%20f'%12.345)

print('%-20f'%12.345)

print('%+020f'%12.345)
